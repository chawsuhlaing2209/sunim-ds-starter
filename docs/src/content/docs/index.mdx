import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Explained here, rendered there" icon="open-book">
    Every component page is generated from that component's own intent file. Storybook
    is where the variants render; this is where they are explained. Each page
    deep-links into the matching stories.
  </Card>
  <Card title="Written once, checked once" icon="approve-check">
    The intent that becomes the *When to use* section is the same file the release
    gate reads. A page cannot be published for a component whose intent would fail
    that gate — the generator refuses.
  </Card>
  <Card title="Nothing here is hand-written" icon="setting">
    Component pages come out of `scripts/generate-docs.mjs`, from the props, the
    tokens, the stories and the intent. Editing one by hand lasts until the next
    build.
  </Card>
  <Card title="It says what it does not promise" icon="warning">
    Accessibility limits, values Figma never bound, and what a `0.x` version
    explicitly does not guarantee. Each page carries them rather than leaving them
    for a consumer to discover.
  </Card>
</CardGrid>
